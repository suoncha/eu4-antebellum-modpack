#TCR - Begin Crisis Information
country_event = {
	id = abf_tcr_yuezhi_crisis.1
	title = abf_tcr_yuezhi_crisis.1.t
	desc = abf_tcr_yuezhi_crisis.1.d
	picture = MONGOL_BANNERS_eventPicture

	major = yes
	
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			# Save ROOT as event target
			ROOT = {
				save_global_event_target_as = abf_tcr_yuezhi_crisis_invader
			}
			# Set global flag for crisis beginning 
			if = {
				limit = {
					NOT = {
						has_saved_global_event_target = abf_tcr_yuezhi_crisis_target
					}	
				}
				set_global_flag = abf_tcr_yuezhi_crisis_begun_flag
			}
			# Save Event Target/Country flag for EOC 
			if = {
				limit = {
					1816 = {
						owner = {
							is_neighbor_of = ROOT
						}
					}
				}
				hidden_effect = {
					1816 = {
						owner = {
							save_global_event_target_as = abf_tcr_yuezhi_crisis_target
						}
					}
				}
			}
			else_if = {
				limit = { 
					event_target:abf_tcr_yuezhi_crisis_target = {
						exists = no
					}
					event_target:EmperorOfChina = {
						is_neighbor_of = ROOT
						owns = 1816
					}
				}
				event_target:EmperorOfChina = {
					save_global_event_target_as = abf_tcr_yuezhi_crisis_target
					set_country_flag = abf_eoc_yuezhi_target
				}
			}
		}
	}


	#Demand their surrender 
	option = {
		name = abf_tcr_yuezhi_crisis.1.c
		custom_tooltip = ABF_YUEZHI_CRISIS_YUEZHI_DEMANDS_TT
		custom_tooltip = AB_EMPTY
		event_target:abf_tcr_yuezhi_crisis_target = {
			country_event = {
				id = abf_tcr_yuezhi_crisis.2
			}
		}
	}
}
