### Preset Colonial Nations ###
namespace = colonial_rome

#Eastern America
country_event = {
	id = colonial_rome.1
	title = colonial_rome.1.t
	desc = colonial_rome.1.d
	picture = COLONIZATION_eventPicture
	
	is_triggered_only = yes
	hidden = yes
	
	trigger = {
		capital_scope = {
			colonial_region = colonial_eastern_america
			is_core = ROOT
		}
		NOT = { exists = RO1 }
	}
	option = {
		name = colonial_rome.a
		change_tag = RO1
		swap_free_idea_group = yes
		swap_non_generic_missions = yes
		hidden_effect = {
			if = {
				limit = {
					culture_group = roman_g
				}
				overlord = {
					country_event = {
						id = colonial_culture_rome.1
					}
				}
			}
		}
	}
}
