	title = abf_zun_event.3.t
	desc = abf_zun_event.3.d

	is_triggered_only = yes

	option = {
		name = abf_zun_event.3.a
		custom_tooltip = ABF_ZNN_GLOBAL_PURIFICATION_CB_TT
		hidden_effect = {
			every_country = {
				limit = {
					NOT = { tag = ROOT } 
					NOT = {
						religion = zunist
					}
				}
				ROOT = {
					add_casus_belli = {
						type = cb_abf_zun_crusade 
						months = 300                
						target = PREV              
					}
				}
			}
		}
	}
}

country_event = {
	id = abf_zun_event.7
	title = abf_zun_event.7.t
	desc = abf_zun_event.7.d
	picture = BORDER_TENSION_eventPicture
	
	is_triggered_only = yes

	option = { #Integrate the Marathas
		name = abf_zun_event.7.a
		custom_tooltip = ABF_ZNN_UNLOCK_MARATHAS_PRIVILEGE_TT
		hidden_effect = {
			unlock_estate_privilege = {
				estate_privilege = estate_nobles_abf_znn_maratha_special_privilege
			}
		}
	}

	option = { #Our current class of warrior-nobility is enough
		name = abf_zun_event.7.b
		custom_tooltip = AB_SHH_improved_nobles_privilege_tt
		hidden_effect = {
			if = {
				limit = {
					has_estate_privilege = estate_nobles_command_of_the_military
				}
				remove_estate_privilege = estate_nobles_command_of_the_military
				set_country_flag = AB_SHH_improved_nobles_privilege
				set_estate_privilege = AB_estate_nobles_command_of_the_military_improved
			}
			else = {
				set_country_flag = AB_improved_command_military_privilege
			}
		}
	}
}
