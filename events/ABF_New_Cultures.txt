## Nikkei (Japanese) Spread [[[
province_event = {
	id = abf_new_cultures.1
	title = abf_new_cultures.1.t
	desc = abf_new_cultures.1.d
	picture = CULTURE_eventPicture
	hidden = yes

	mean_time_to_happen = {
		months = 240
		modifier = {
            factor = 0.7
            owner = {
                has_country_flag = nci_speed_up_cultures
            }
        }
	}

#Will fire for all non japanese_g province in the philippines
trigger = {
	owner = {
		OR = {
			NOT = {
				tag = RMR 
			}
			NOT = {
				tag = ROM
			}
		}
	}
	owner = {
        has_country_flag = abf_nikkei_culture_spread
    }
	NOT = {
		owner = {
			has_country_flag = abf_japan_embraced_chinese_culture
		}
	}
	religion = owner
    NOT = {
        culture_group = japanese_g
    }
    OR = {
        area = palawan_area
        area = luzon_area
        area = visayas_area
        area = west_mindanao_area
        area = mindanao_area
        area = southern_luzon_area
    }
}
	
	option = {
		name = abf_new_cultures.1.a	
		change_culture = abf_japanese_nikkei
	}
}
