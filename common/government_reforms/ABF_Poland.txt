#----Crown of Poland with Popular Sovereignity Mechanic----#

	icon = "integrated_sejmiks_reform"
	allow_convert = no
	allow_normal_conversion = no
	lock_level_when_selected = yes
	legacy_equivalent = feudal_monarchy

	potential = {
		OR = {
			tag = PLN
			was_tag = PLN
		}
		custom_trigger_tooltip = {
			tooltip = AB_EMPTY
			has_country_flag = abf_pln_crownpoland_popular
		}
	}

	trigger = {
		if = {
			limit = {
				custom_trigger_tooltip = {
					tooltip = AB_EMPTY
					has_country_flag = abf_pln_crownpoland_popular
				}
				AND = {
					has_reform = abf_crownpoland
					OR = {
						tag = POL
						tag = PLC
					}
				}
			}
		}
		else = {
			custom_trigger_tooltip = {
				tooltip = AB_CROWN_POLAND_VIA_DECISION
				always = no
			}
		}
	}
	
	valid_for_nation_designer = yes
	nation_designer_cost = 0
	modifiers = {
		legitimacy = 1
		global_unrest = -1
	}

	}
}
